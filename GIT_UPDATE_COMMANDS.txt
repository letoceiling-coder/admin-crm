  КОМАНДЫ ДЛЯ ОБНОВЛЕНИЯ ФАЙЛОВ ИЗ GIT НА СЕРВЕРЕ
═══════════════════════════════════════════════════════════════

┌─────────────────────────────────────────────────────────────┐
│ ВАРИАНТ 1: Через Terminal в панели управления хостингом     │
└─────────────────────────────────────────────────────────────┘

1. Войдите в панель управления хостингом
2. Откройте "Terminal" или "SSH Access"
3. Выполните команды:

cd ~/crm.neeklo.ru/public_html

# Проверка текущего статуса
git status

# Проверка remote репозитория
git remote -v

# Если remote не настроен, добавьте:
git remote add origin https://github.com/letoceiling-coder/admin-crm.git

# Получить последние изменения из репозитория
git fetch origin

# Проверить текущую ветку
git branch

# Переключиться на main (если нужно)
git checkout main
# или
git checkout -b main

# Обновить файлы из репозитория (жесткий сброс)
git reset --hard origin/main

# Или если есть локальные изменения, которые нужно сохранить:
git stash
git pull origin main
git stash pop

# После обновления выполнить:
composer install --no-dev --optimize-autoloader
php artisan migrate --force
php artisan config:clear
php artisan cache:clear
php artisan route:clear
php artisan view:clear

┌─────────────────────────────────────────────────────────────┐
│ ВАРИАНТ 3: Безопасное обновление (с сохранением изменений)  │
└─────────────────────────────────────────────────────────────┘

cd ~/crm.neeklo.ru/public_html

# Сохранить локальные изменения
git stash save "Backup before update $(date +%Y-%m-%d_%H-%M-%S)"

# Получить обновления
git fetch origin

# Обновить файлы
git reset --hard origin/main

# Если нужно вернуть локальные изменения:
# git stash pop


┌─────────────────────────────────────────────────────────────┐
│ ВАРИАНТ 4: Через deploy команду (рекомендуется)             │
└─────────────────────────────────────────────────────────────┘

На локальной машине:

cd c:\OSPanel\domains\ADMIN-CRM\crm

# Сначала закоммитить и отправить изменения
git add .
git commit -m "Обновление проекта"
git push origin main

# Затем выполнить deploy (автоматически обновит сервер)
php artisan deploy

# Или без сборки фронтенда
php artisan deploy --skip-build

Проблема: "fatal: not a git repository"
Решение:
  git init
  git remote add origin https://github.com/letoceiling-coder/admin-crm.git

Проблема: "error: The following untracked working tree files would be overwritten"
Решение:
  git stash
  git reset --hard origin/main

Проблема: "Permission denied"
Решение:
  chmod -R 775 storage bootstrap/cache
  chown -R dsc23ytp:dsc23ytp .

Проблема: "git safe directory"
Решение:
  git config --global --add safe.directory ~/crm.neeklo.ru/public_html

Проблема: "composer: command not found"
Решение:
  # Использовать полный путь или установить composer
  /usr/local/bin/composer install --no-dev --optimize-autoloader
  # или
  php composer.phar install --no-dev --optimize-autoloader
